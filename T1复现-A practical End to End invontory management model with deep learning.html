一、复现框架（简要）

01 / 库存环境与数据

生成历史需求、供应商交货提前期、SKU类别、期初库存及未来真实需求；考虑持有与缺货成本。

02 / 最优补货标签

根据论文定理1，在相邻到货期间的累计需求路径上使用临界分位数，计算事后最优补货量。

03 / E2E神经网络

需求GRU编码器、VLT多层感知机、融合网络直接输出补货量；两个辅助输出预测需求和提前期。

04 / 训练与对照

用补货量标签监督训练，并与事后最优、正态/Gamma基础库存规则和简单点预测策略比较。

05 / 评估与绘图

计算样本外库存持有、缺货与总成本；绘制结构图、需求预测图、成本图、学习曲线和敏感性图。

复现范围：使用合成数据，未使用京东真实数据；GRU是对原文MQRNN的简化替代，优化器采用Adam而非原文附录的SGD配置。原文的现场实验、倾向得分匹配和DID不能凭合成数据精确复现。以下所有图表均为模拟结果，不是论文原图。

Run: python reproduce.py --epochs 35 --n-train 1600 --n-test 400
Not the proprietary JD.com data or the exact original MQRNN implementation.
"""

from pathlib import Path

import torch

import matplotlib

def make_data(n, rng, history=21, horizon=12, review=7, h=1., b=9.):
    """Simulated SKU demand with promotions, seasonality, and vendor lead-time heterogeneity."""
    hist, vhist, basic, stock, future, arrivals, labels, dmean, lmean = [], [], [], [], [], [], [], [], []
    for _ in range(n):
        category = int(rng.integers(0, 3))
        base = rng.uniform(5, 22) * (1 + 0.2 * category)
        trend = rng.uniform(-.18, .18)
        promo = rng.uniform(0, .35)
        weekday = rng.uniform(0, 2 * np.pi)
        noise = rng.uniform(.12, .35)
        t = np.arange(history + horizon)
        seasonal = 1 + .14 * np.sin(2 * np.pi * t / 7 + weekday)
        multiplier = 1 + trend * (t / (history + horizon))
        demand_mean = base * seasonal * multiplier * (1 + promo * (t % 9 == 0))
        draws = rng.poisson(np.maximum(demand_mean, .5)).astype(float)
        demand_hist = draws[:history]
        future_dem = draws[history:]
        vlt_base = 2 + category * .65 + rng.uniform(0, 1.0)
        vlt_hist = np.clip(np.rint(vlt_base + rng.normal(0, .65, 8)), 1, 6)
        lead = int(np.clip(np.rint(vlt_base + rng.normal(0, .8)), 1, 5))
        next_lead = int(np.clip(np.rint(vlt_base + rng.normal(0, .8)), 1, 5))
        arrival = lead
        next_arrival = min(horizon, review + next_lead)
        if next_arrival <= arrival:
            next_arrival = min(horizon, arrival + 1)
        # I_v: stock observed immediately before current order arrives.
        inventory_at_arrival = max(0., rng.normal(base * 2.8, base * 1.0))
        # Theorem 1: critical fractile over cumulative demands at each day in [v_m, v_{m+1}).
        cumulative = np.cumsum(future_dem[arrival:next_arrival])
        idx = min(len(cumulative)-1, int(np.ceil(b/(h+b) * len(cumulative))) - 1)
        target = max(float(cumulative[idx]) - inventory_at_arrival, 0.)
        hist.append(demand_hist)
        vhist.append(vlt_hist)
        basic.append([float(category == 0), float(category == 1), float(category == 2), promo, noise])
        stock.append(inventory_at_arrival)
        future.append(future_dem)
        arrivals.append([arrival, next_arrival])
        labels.append(target)
        dmean.append(float(future_dem.mean()))
        lmean.append(float(lead))
    return dict(hist=np.asarray(hist,np.float32), vhist=np.asarray(vhist,np.float32),
                basic=np.asarray(basic,np.float32), stock=np.asarray(stock,np.float32),
                future=np.asarray(future,np.float32), arrivals=np.asarray(arrivals,np.int32),
                label=np.asarray(labels,np.float32), demand_target=np.asarray(dmean,np.float32),
                lead_target=np.asarray(lmean,np.float32))


class E2ENet(nn.Module):
    """Modular GRU demand encoder + VLT MLP + joint replenishment head.
    GRU is a simplified stand-in for the paper's MQRNN, not an exact reimplementation.
    """
    def __init__(self):
        super().__init__()
        self.demand = nn.GRU(input_size=1, hidden_size=50, batch_first=True)
        self.vlt = nn.Sequential(nn.Linear(8,50), nn.ReLU(), nn.Linear(50,20), nn.ReLU())
        self.aux_d = nn.Linear(50, 1)
        self.aux_l = nn.Linear(20, 1)
        self.integration = nn.Sequential(nn.Linear(50+20+5+2,100), nn.ReLU(),
                                         nn.Dropout(.10), nn.Linear(100,100), nn.ReLU(),
                                         nn.Linear(100,1), nn.Softplus())
    def forward(self, hist, vhist, basic, stock):
        _, hidden = self.demand(hist.unsqueeze(-1) / 30.)
        d = hidden[-1]
        l = self.vlt(vhist / 6.)
        merged = torch.cat([d,l,basic,torch.full_like(stock, 7./10.),stock/100.],dim=1)
        order = self.integration(merged) * 100.
        return order.squeeze(1), self.aux_d(d).squeeze(1)*30., self.aux_l(l).squeeze(1)*6.


def tensorize(data):
    return tuple(torch.from_numpy(data[k]) for k in ['hist','vhist','basic','stock','label','demand_target','lead_target'])


def train(model, train_data, test_data, epochs=18, batch=64, seed=42):
    torch.manual_seed(seed)
    opt = torch.optim.Adam(model.parameters(),lr=.001)
    # Adam is a practical substitution for the paper's SGD+momentum configuration; scaled loss stabilizes training.
    loader = DataLoader(TensorDataset(*tensorize(train_data)),batch_size=batch,shuffle=True)
    history=[]
    for ep in range(epochs):
        model.train()
        running=0.
        for h,v,b,s,y,dy,ly in loader:
            p,pd,pl=model(h,v,b,s.unsqueeze(1))
            loss=((p-y)/100.).square().mean()+.1*((pd-dy)/30.).square().mean()+.1*((pl-ly)/6.).square().mean()
            opt.zero_grad(); loss.backward(); nn.utils.clip_grad_norm_(model.parameters(),5.); opt.step()
            running += loss.item()*len(y)
        model.eval()
        with torch.no_grad():
            h,v,b,s,y,dy,ly=tensorize(test_data)
            p,pd,pl=model(h,v,b,s.unsqueeze(1))
            val=(((p-y)/100.).square().mean()+.1*((pd-dy)/30.).square().mean()+.1*((pl-ly)/6.).square().mean()).item()
        history.append((ep+1,running/len(train_data['label']),val))
        print(f'Epoch {ep+1:02d}: train={history[-1][1]:.4f} validation={val:.4f}')
    return np.asarray(history)


def predict(model, data):
    model.eval()
    with torch.no_grad():
        h,v,b,s,*_=tensorize(data)
        p,pd,pl=model(h,v,b,s.unsqueeze(1))
    return p.numpy(),pd.numpy(),pl.numpy()


def policies(data, e2e):
    h,b=1.,9.
    N=len(e2e)
    methods={k:np.zeros(N) for k in ['OPT (oracle)','E2E (GRU)','PTO-normal','PTO-gamma','PTO-point']}
    methods['OPT (oracle)'][:]=data['label']
    methods['E2E (GRU)'][:]=e2e
    for i in range(N):
        hist=data['hist'][i]; vhist=data['vhist'][i]
        inventory=data['stock'][i]
        R=7.
        mean=max(float(hist.mean()),.1)
        sd=max(float(hist.std()),1.)
        lead=max(float(vhist.mean()),1.)
        lead_sd=max(float(vhist.std()),.2)
        # Paper Eq. (7) normal base-stock approximation; subtract stock at arrival here.
        S=mean*(R+lead)+norm.ppf(b/(b+h))*np.sqrt((R+lead)*sd**2+mean**2*lead_sd**2)
        methods['PTO-normal'][i]=max(0.,S-inventory)
        shape=max((mean/sd)**2,.05); scale=max(sd**2/mean,.05)
        methods['PTO-gamma'][i]=max(0.,gamma.ppf(b/(b+h),a=(R+lead)*shape,scale=scale)-inventory)
        methods['PTO-point'][i]=max(0.,mean*R-inventory)
    costs={}; holds={}; shorts={}
    for name,orders in methods.items():
        hold=np.zeros(N); short=np.zeros(N)
        for i in range(N):
            start,end=data['arrivals'][i]
            path=np.cumsum(data['future'][i,start:end])
            remaining=data['stock'][i]+max(0.,orders[i])-path
            hold[i]=np.maximum(remaining,0.).sum()*h
            short[i]=np.maximum(-remaining,0.).sum()*b
        costs[name]=float(np.mean(hold+short))
        holds[name]=float(hold.mean()); shorts[name]=float(short.mean())
    return costs,holds,shorts

def main():
    parser=argparse.ArgumentParser()
    parser.add_argument('--epochs',type=int,default=35)
    parser.add_argument('--n-train',type=int,default=1600)
    parser.add_argument('--n-test',type=int,default=400)
    parser.add_argument('--seed',type=int,default=42)
    args=parser.parse_args()
    torch.manual_seed(args.seed);np.random.seed(args.seed)
    train_data=make_data(args.n_train,np.random.default_rng(args.seed))
    test_data=make_data(args.n_test,np.random.default_rng(args.seed+1))
    model=E2ENet()
    for m in model.modules():
        if isinstance(m,nn.Linear):
            nn.init.normal_(m.weight,mean=0.,std=.01)
            nn.init.zeros_(m.bias)
    loss=train(model,train_data,test_data,epochs=args.epochs,seed=args.seed)
    orders,dp,lp=predict(model,test_data)
    costs,holds,shorts=policies(test_data,orders)
    plots(test_data,model,loss,costs,holds,shorts,dp)
    import csv
    with open(OUT/'cost_summary.csv','w',newline='') as f:
        wr=csv.writer(f);wr.writerow(['method','holding','stockout','total'])
        for name in costs:wr.writerow([name,round(holds[name],4),round(shorts[name],4),round(costs[name],4)])
    np.savetxt(OUT/'training_loss.csv',loss,delimiter=',',header='epoch,train_loss,val_loss',comments='')
    torch.save(model.state_dict(),OUT/'model_weights.pt')
    print('Saved outputs in:',OUT)
    print('Mean costs:',{k:round(v,2) for k,v in costs.items()})

if __name__=='__main__':main()
