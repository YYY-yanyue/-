"""Yan et al. (2026) T2: illustrative, executable small-scale reproduction.
Four policies: LIR, WNH, GDSH, state-dependent teacher-constrained DRL.
Not the paper's 432-case exact numerical replication.
"""
import argparse, json, itertools, math
from pathlib import Path
import numpy as np
import torch
from torch import nn
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
from scipy.stats import norm

torch.set_num_threads(2)
ROOT=Path(__file__).resolve().parent

class PolicyNet(nn.Module):
    def __init__(self, L, p, delta):
        super().__init__()
        self.layers=nn.Sequential(nn.Linear(L+1,8),nn.ReLU(),nn.Linear(8,16),nn.ReLU(),nn.Linear(16,16),nn.ReLU(),nn.Linear(16,2*(2**(L-1))))
        self.L,self.delta=L,delta
        self.register_buffer('teacher', teacher_weights(L,p))
        # Center the initial output at the WNH teacher policy.
        nn.init.zeros_(self.layers[-1].weight)
        nn.init.zeros_(self.layers[-1].bias)
    def forward(self,s):
        logits=self.layers(s).reshape(-1,2**(self.L-1),2)
        eta=torch.softmax(logits,dim=-1)
        return eta[...,0]*(self.teacher-self.delta)+eta[...,1]*(self.teacher+self.delta)

def yield_paths(L):
    return torch.tensor(list(itertools.product([0.,1.],repeat=L-1)),dtype=torch.float32)

def teacher_weights(L,p):
    paths=yield_paths(L)
    return (p**paths.sum(1))*((1-p)**((L-1)-paths.sum(1)))

def target_stock(L,mu,sigma,p,ch,cb):
    # Paper Eq. (3): rho_Y^2=(1-p)/p. For p>0.5 denominator stays positive.
    rho2=(1-p)/p
    k=norm.ppf(cb/(cb+ch))
    return (L+1)*mu+k*math.sqrt((L+1)*sigma**2+max(L,1)*rho2/(1-rho2)*(mu**2+sigma**2))

def noise(seed,B,T,mu,sigma,p):
    rng=np.random.default_rng(seed)
    demand=np.maximum(0,rng.normal(mu,sigma,(B,T))).astype('float32')
    yield_ok=(rng.random((B,T))<p).astype('float32')
    return torch.from_numpy(demand),torch.from_numpy(yield_ok)

def rollout(kind, demand, yield_ok, cfg, net=None, free_logits=None, record=False):
    B,T=demand.shape; L=cfg['L']; p=cfg['p']; S=cfg['S']; paths=yield_paths(L)
    teacher=teacher_weights(L,p)
    I=torch.zeros(B); pipeline=torch.zeros(B,L)  # first entry arrives at period start
    cost=torch.zeros(B); discount=1.; orders=[]; levels=[]; step_costs=[]
    for t in range(T):
        arriving=pipeline[:,0]*yield_ok[:,t]
        I=I+arriving
        active=pipeline[:,1:]  # q_(t-L+1),...,q_(t-1)
        # Each row is one possible future yield pattern for the in-transit orders.
        ideals=torch.relu(S-I[:,None]-active@paths.T)
        if kind=='LIR':
            q=torch.relu(S-I-p*active.sum(1))
        elif kind=='WNH':
            q=ideals@teacher
        elif kind=='GDSH':
            q=ideals@free_logits
        elif kind=='DRL':
            state=torch.cat((torch.full((B,1),S),I[:,None],active),dim=1)
            # Normalize scale for stable NN training; state content unchanged.
            q=(ideals*net(state/(S+1e-6))).sum(1)
        else: raise ValueError(kind)
        # Final L periods: orders cannot arrive within the finite horizon.
        if t>=T-L: q=q*0
        ending=I-demand[:,t]
        one=cfg['co']*arriving+cfg['ch']*torch.relu(ending)+cfg['cb']*torch.relu(-ending)
        cost=cost+discount*one
        discount*=cfg['alpha']
        pipeline=torch.cat((active,q[:,None]),dim=1)
        I=ending
        if record:
            orders.append(q.detach().mean().item());levels.append(I.detach().mean().item());step_costs.append(one.detach().mean().item())
    if record: return cost,{'orders':orders,'inventory':levels,'period_cost':step_costs}
    return cost

def main():
    ap=argparse.ArgumentParser()
    ap.add_argument('--epochs',type=int,default=45)
    ap.add_argument('--train-episodes',type=int,default=96)
    ap.add_argument('--test-episodes',type=int,default=100)
    ap.add_argument('--T',type=int,default=35)
    ap.add_argument('--L',type=int,default=3)
    args=ap.parse_args()
    torch.manual_seed(7);np.random.seed(7)
    cfg={'L':args.L,'T':args.T,'mu':10.,'sigma':2.,'p':.75,'co':2.,'ch':1.,'cb':9.,'alpha':.99}
    cfg['S']=target_stock(args.L,cfg['mu'],cfg['sigma'],cfg['p'],cfg['ch'],cfg['cb'])
    cfg['delta']=.25 if args.L in (2,3) else .0884
    tr_d,tr_y=noise(101,args.train_episodes,args.T,cfg['mu'],cfg['sigma'],cfg['p'])
    te_d,te_y=noise(202,args.test_episodes,args.T,cfg['mu'],cfg['sigma'],cfg['p'])
    with torch.no_grad(): teacher_cost=rollout('WNH',tr_d,tr_y,cfg)
    # GDSH: fixed, unconstrained state-independent scenario weights.
    w=nn.Parameter(teacher_weights(args.L,cfg['p']).clone())
    opt_g=torch.optim.Adam([w],lr=.003)
    # DRL: state-dependent weights constrained within teacher +/- delta.
    net=PolicyNet(args.L,cfg['p'],cfg['delta'])
    opt_d=torch.optim.Adam(net.parameters(),lr=.003)
    gh=[];dh=[]
    for epoch in range(args.epochs):
        lr=max(.0001,.003*.98**epoch)
        for opt in (opt_g,opt_d):
            for group in opt.param_groups:group['lr']=lr
        opt_g.zero_grad()
        cg=rollout('GDSH',tr_d,tr_y,cfg,free_logits=w)
        lg=(cg/teacher_cost).mean()
        lg.backward();torch.nn.utils.clip_grad_norm_([w],10);opt_g.step()
        # No nonnegativity restriction for GDSH weights in Eq. (5);
        # enforce nonnegative final order quantity to respect q>=0.
        with torch.no_grad():w.clamp_(min=0)
        opt_d.zero_grad()
        cd=rollout('DRL',tr_d,tr_y,cfg,net=net)
        rel=(cd-teacher_cost)/(teacher_cost+1e-6)
        ld=torch.exp((100*rel).clamp(-20,20)).mean()
        ld.backward();torch.nn.utils.clip_grad_norm_(net.parameters(),10);opt_d.step()
        gh.append(cg.detach().mean().item());dh.append(cd.detach().mean().item())
    results={};records={}
    with torch.no_grad():
        for kind in ['LIR','WNH','GDSH','DRL']:
            costs,record=rollout(kind,te_d,te_y,cfg,net=net,free_logits=w,record=True)
            results[kind]={'mean':round(costs.mean().item(),3),'std':round(costs.std().item(),3)}
            records[kind]=record
    figures=ROOT/'figures';figures.mkdir(exist_ok=True)
    plt.rcParams.update({'font.size':10,'figure.dpi':130})
    fig,ax=plt.subplots(figsize=(7.5,4));ax.bar(list(results),[v['mean'] for v in results.values()]);ax.set_title('Held-out test cost (100 common trajectories)');ax.set_ylabel('Discounted total cost');fig.tight_layout();fig.savefig(figures/'01_cost_comparison.png');plt.close(fig)
    fig,ax=plt.subplots(figsize=(7.5,4));ax.plot(gh,label='GDSH');ax.plot(dh,label='DRL');ax.set_title('Training mean cost (fixed simulation batch)');ax.set_xlabel('Epoch');ax.set_ylabel('Mean discounted cost');ax.legend();fig.tight_layout();fig.savefig(figures/'02_training.png');plt.close(fig)
    fig,ax=plt.subplots(figsize=(7.5,4));
    for kind in results:ax.plot(records[kind]['orders'],label=kind)
    ax.set_title('Mean order quantity by period');ax.set_xlabel('Period');ax.set_ylabel('Order units');ax.legend(ncol=4);fig.tight_layout();fig.savefig(figures/'03_orders.png');plt.close(fig)
    fig,ax=plt.subplots(figsize=(7.5,4));
    for kind in results:ax.plot(records[kind]['inventory'],label=kind)
    ax.set_title('Mean ending inventory by period');ax.set_xlabel('Period');ax.set_ylabel('Inventory (negative = backlog)');ax.legend(ncol=4);fig.tight_layout();fig.savefig(figures/'04_inventory.png');plt.close(fig)
    fig,ax=plt.subplots(figsize=(7.5,4));
    for kind in results:ax.plot(records[kind]['period_cost'],label=kind)
    ax.set_title('Mean one-period cost');ax.set_xlabel('Period');ax.set_ylabel('Cost');ax.legend(ncol=4);fig.tight_layout();fig.savefig(figures/'05_period_cost.png');plt.close(fig)
    report={'configuration':cfg,'train_episodes':args.train_episodes,'test_episodes':args.test_episodes,'epochs':args.epochs,'results':results,'gdsh_weights':[round(x,5) for x in w.detach().tolist()],'training_history':{'GDSH':gh,'DRL':dh},'paper_reference_average_cost':{'LIR':4101,'WNH':4076,'GDSH':3960,'DRL':3953}}
    (ROOT/'results.json').write_text(json.dumps(report,ensure_ascii=False,indent=2),encoding='utf-8')
    print(json.dumps(results,ensure_ascii=False,indent=2))

if __name__=='__main__':main()

02｜模型与状态转移

每一期先验收到提前 L 期订购的货，质量检验以概率 p 整单成功，随后观察库存及在途订单并下单，再发生需求。未满足需求延期交付，负库存代表欠货。

状态：st = (It, Qt)；动作：qt ≥ 0；
库存更新：It+1 = It + Yt+1qt+1−L − Dt
单期成本：ct = coYtqt−L + ch[It−Dt]+ + cb[Dt−It]+
优化目标：min E[Σt=1T αt−1ct]

本次参数：L=3、T=20、需求均值 μ=10.0、标准差 σ=2.0、成功率 p=0.75、订货成本 co=2.0、持有成本 ch=1.0、欠货成本 cb=9.0、折现因子 α=0.99；按论文式(3)计算目标库存 SL≈56.81。

03｜四种补货策略怎么复现

LIR：使用当前库存与在途订单的期望成功量，补足至目标库存。按原文选择通胀系数 β=1。

qtLIR = [SL − It − p ΣQt]+

WNH：对 L−1 个在途订单枚举 2L−1 种成功/失败组合，计算每个组合下的缺口，再以该组合的伯努利概率加权。

gj(st) = [SL − It − Σiyijqi]+；qtWNH = ΣjP(j)gj(st)

GDSH：保留WNH的情景缺口，使用梯度下降学习一组所有状态共用的权重 wj，以整个时间跨度的成本为优化目标。

qtGDSH = Σjwjgj(st)

DRL：将 SL、当前库存和在途订单输入全连接网络（隐藏层8→16→16），输出 2L−1×2 个logit，逐行Softmax后将各情景权重限制在WNH权重 ±δ 范围内，再计算订货量。训练不是DQN或TD误差，而是沿完整模拟轨迹对折现成本进行反向传播。

wjt = ηjt(wjWNH−δ) + (1−ηjt)(wjWNH+δ)
qtDRL = Σjwjtgj(st)
损失：ℒ = exp(100 × (CTDRL − CTWNH)/CTWNH)

论文第5–7页给出WNH/GDSH/DRL公式及网络示意（Fig.1、Fig.2），第8页给出训练算法和参数；实现中对数值溢出做了梯度裁剪与指数截断。

## 关键实现

- 需求为截断到非负的正态分布；整单交货成功服从 Bernoulli(p)；未满足需求延期交付。
- 订货成本仅在到货成功时计入，库存成本在需求发生后计入，采用折现总成本。
- LIR 和 WNH 是两种基准；GDSH 学习全局固定情景权重；DRL 使用三隐藏层8/16/16、逐行Softmax和 WNH ±δ 权重范围。
- 训练用共享模拟样本路径，测试使用不同随机种子和共享测试路径。
- 论文正文称输入维度 L+1，附录A称 L；本项目采用包含 S_L、I 和 L−1 个在途订单的 L+1 维表示。
- 为节约资源，演示版训练12轮、48条训练路径、T=20、L=3；论文报告300轮、500条训练路径、100条测试路径。
- GDSH 在实现中对权重做非负截断，以保证订货量非负；这是数值稳定性选择，可能与原文实现细节不同。
- 由于有限样本与短训练，DRL不一定优于GDSH；不能将结果视作论文表2复现。
