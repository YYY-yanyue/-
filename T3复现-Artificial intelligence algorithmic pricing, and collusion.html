复现框架（简要）

本项目按“构建市场 → 定义状态与动作 → 双智能体学习 → 检验训练结果 → 绘制论文图形”的顺序复现论文的基准定价实验。

  1. 构建竞争市场：设置两家企业、Logit需求函数与单位边际成本；先计算静态Bertrand-Nash价格和联合利润最大化价格，作为后续比较基准。
  2. 定义强化学习问题：每家企业从15个离散价格中选择动作；状态为上一期两家企业的价格组合，因此每个智能体有225种状态、15种动作。
  3. 进行Q-learning训练：两家企业同时定价，分别得到当期利润；按照 Q ← Q + α[利润 + δ×下一状态最大Q值 − Q] 更新各自的Q表，并通过随时间递减的ε-greedy策略探索。
  4. 评估学习结果：观察贪心策略是否稳定，计算收敛后（或训练截止时）的平均利润，并以竞争利润与联合利润最大化利润进行标准化，得到利润增益Δ。
  5. 生成图形：改变学习率、探索率和折扣因子；再人为让一家企业短暂降价，观察另一家企业的反应和后续价格恢复，绘制Figure 1、3、4、5、10的近似复现图。

说明：快速模式使用缩小的实验规模；部分图形为教学性近似复现，不等同于论文的完整统计结果。

完整 Python 源代码

"""Educational reproduction of Calvano et al. (2020), AER.
Run: python reproduce.py --mode quick [--seed 2026]
     python reproduce.py --mode paper  # computationally expensive
Outputs: results/*.png, results/summary.csv
"""
import argparse
from pathlib import Path
import numpy as np
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
from scipy.optimize import brentq, minimize_scalar
from numba import njit

OUT = Path(__file__).resolve().parent / 'results'
OUT.mkdir(exist_ok=True)

# Baseline: c=1, a=2, a0=0, mu=.25, n=2, m=15, xi=.1, k=1.
C, A, A0, MU, M, XI = 1., 2., 0., .25, 15, .1

def demand(p, other):
    v=np.exp((A-p)/MU); u=np.exp((A-other)/MU)
    return v/(v+u+np.exp(A0/MU))

def profit(p, other):
    return (p-C)*demand(p,other)

def foc(p, other):
    d=demand(p,other)
    return 1-(p-C)*(1-d)/MU

p_n=brentq(lambda p:foc(p,p),1.01,2.5)
p_m=minimize_scalar(lambda p:-2*profit(p,p),bounds=(p_n,3),method='bounded').x
prices=np.linspace(p_n-XI*(p_m-p_n),p_m+XI*(p_m-p_n),M)
pay=np.array([[profit(x,y) for y in prices] for x in prices])
pi_n=profit(p_n,p_n); pi_m=profit(p_m,p_m)

@njit
def argmax_low(row):
    best=0
    for j in range(1,len(row)):
        if row[j]>row[best]+1e-12: best=j
    return best

@njit
def train(pay, alpha, beta, discount, steps, patience, seed, record_every):
    np.random.seed(seed)
    m=pay.shape[0]; ns=m*m
    q=np.zeros((2,ns,m),dtype=np.float64)
    # Equation (8): expected per-period profit vs uniformly randomized rival / (1-discount)
    for a in range(m):
        v=0.
        for b in range(m): v+=pay[a,b]
        v/=m*(1-discount)
        for s in range(ns):
            q[0,s,a]=v; q[1,s,a]=v
    prev0=np.random.randint(m);prev1=np.random.randint(m)
    state=prev0*m+prev1
    oldpolicy=np.zeros((2,ns),dtype=np.int64)
    for i in range(2):
        for s in range(ns):oldpolicy[i,s]=argmax_low(q[i,s])
    unchanged=0; stop=steps
    nrecord=(steps+record_every-1)//record_every
    history=np.full((nrecord,3),np.nan)
    k=0; sumgain=0.;count=0
    for t in range(steps):
        eps=np.exp(-beta*t)
        a0=argmax_low(q[0,state]);a1=argmax_low(q[1,state])
        if np.random.random()<eps:a0=np.random.randint(m)
        if np.random.random()<eps:a1=np.random.randint(m)
        r0=pay[a0,a1];r1=pay[a1,a0]
        nxt=a0*m+a1
        v0=q[0,nxt,argmax_low(q[0,nxt])]
        v1=q[1,nxt,argmax_low(q[1,nxt])]
        q[0,state,a0]+=alpha*(r0+discount*v0-q[0,state,a0])
        q[1,state,a1]+=alpha*(r1+discount*v1-q[1,state,a1])
        state=nxt
        sumgain+=(r0+r1)*.5;count+=1
        if (t+1)%record_every==0:
            history[k,0]=t+1;history[k,1]=sumgain/count;history[k,2]=eps
            k+=1;sumgain=0.;count=0
        # Greedy-policy convergence check: exact all-state policy unchanged for patience steps.
        # Checking every step is cheap for m=15 but costly; check every 100 steps.
        if (t+1)%100==0:
            changed=False
            for i in range(2):
                for s in range(ns):
                    x=argmax_low(q[i,s])
                    if x!=oldpolicy[i,s]:changed=True;oldpolicy[i,s]=x
            if changed:unchanged=0
            else:unchanged+=100
            if unchanged>=patience:
                stop=t+1
                break
    policy=np.zeros((2,ns),dtype=np.int64)
    for i in range(2):
        for s in range(ns):policy[i,s]=argmax_low(q[i,s])
    return policy, q, history[:k],stop,state

@njit
def greedy_rollout(policy, pay, state, n=300):
    m=pay.shape[0]
    r=0.; prices_out=np.empty((n,2),dtype=np.int64)
    for t in range(n):
        a=policy[0,state];b=policy[1,state]
        prices_out[t,0]=a;prices_out[t,1]=b
        r+=(pay[a,b]+pay[b,a])*.5
        state=a*m+b
    return r/n,prices_out,state

@njit
def impulse(policy,pay,start,periods=16):
    m=pay.shape[0]
    baseline=np.empty((periods+1,2),dtype=np.int64)
    deviation=np.empty((periods+1,2),dtype=np.int64)
    st=start
    for t in range(periods+1):
        a=policy[0,st];b=policy[1,st]
        baseline[t,0]=a;baseline[t,1]=b;st=a*m+b
    st=start
    for t in range(periods+1):
        a=policy[0,st];b=policy[1,st]
        if t==1:
            best=0
            for j in range(1,m):
                if pay[j,b]>pay[best,b]:best=j
            a=best
        deviation[t,0]=a;deviation[t,1]=b;st=a*m+b
    return baseline,deviation

def run_sessions(alpha,beta,discount,n_sessions,steps,patience,seed,keep=False):
    gains=[];stops=[];trajectories=[];policies=[];paths=[];impulses=[]
    for j in range(n_sessions):
        policy,q,h,stop,state=train(pay,alpha,beta,discount,steps,patience,seed+j*7919,2000)
        avg,acts,end=greedy_rollout(policy,pay,state,400)
        gains.append((avg-pi_n)/(pi_m-pi_n));stops.append(stop)
        if keep:
            trajectories.append(h)
            policies.append(policy)
            paths.append(acts)
            base,dev=impulse(policy,pay,end,16)
            impulses.append((base,dev))
    if keep:return np.array(gains),np.array(stops),trajectories,policies,paths,impulses
    return np.array(gains),np.array(stops)

def plot_main(alpha,beta,n_sessions,steps,patience,seed):
    gains,stops,hists,policies,paths,imps=run_sessions(alpha,beta,.95,n_sessions,steps,patience,seed,True)
    # Figure 10 analogue: average learning path, rolling 100 samples from 2000-step blocks.
    minlen=min(len(h) for h in hists)
    if minlen:
        x=hists[0][:minlen,0]
        y=np.mean(np.array([h[:minlen,1] for h in hists]),axis=0)
        plt.figure(figsize=(8,4.7));plt.plot(x,(y-pi_n)/(pi_m-pi_n),label='Q-learning, sampled learning path')
        plt.axhline(0,color='gray',ls='--',label='Nash profit gain = 0')
        plt.xlabel('Training iterations');plt.ylabel('Normalized profit gain Δ')
        plt.title('Figure 10 analogue: Learning dynamics');plt.legend();plt.tight_layout()
        plt.savefig(OUT/'fig10_learning.png',dpi=170);plt.close()
    # Figure 4 and Figure 5 analogue: compare baseline vs unilateral best response in t=1
    devs=np.array([prices[d] for b,d in imps]);bases=np.array([prices[b] for b,d in imps])
    t=np.arange(devs.shape[1]);mean=devs.mean(axis=0)
    plt.figure(figsize=(8,4.7));plt.plot(t,mean[:,0],'-o',ms=3,label='Deviating firm')
    plt.plot(t,mean[:,1],'-s',ms=3,label='Nondeviating firm')
    plt.axhline(p_n,color='gray',ls='--',label='Nash price')
    plt.axhline(p_m,color='gray',ls=':',label='Monopoly price')
    plt.axhline(bases[:,0,0].mean(),color='black',ls='-.',label='Pred. long-run price')
    plt.xlabel('Time after forced deviation (deviation at t=1)');plt.ylabel('Price')
    plt.title('Figure 4 analogue: Price response to deviation');plt.legend(fontsize=8)
    plt.tight_layout();plt.savefig(OUT/'fig4_impulse.png',dpi=170);plt.close()
    fig,axs=plt.subplots(1,2,figsize=(10,4),sharey=True)
    for i,ax in enumerate(axs):
        dif=devs[:,:,i]-bases[:,:,i]
        ax.plot(t,np.mean(dif,axis=0),color='black',label='Mean')
        ax.fill_between(t,np.percentile(dif,25,axis=0),np.percentile(dif,75,axis=0),alpha=.25,label='25-75%')
        ax.plot(t,dif.min(axis=0),'--',lw=.8,color='gray')
        ax.plot(t,dif.max(axis=0),'--',lw=.8,color='gray')
        ax.set_title(['Deviating firm','Nondeviating firm'][i]);ax.set_xlabel('Time')
    axs[0].set_ylabel('Price change vs no deviation');axs[0].legend(fontsize=8)
    fig.suptitle('Figure 5 analogue: Distribution of responses')
    fig.tight_layout();fig.savefig(OUT/'fig5_impulse_distribution.png',dpi=170);plt.close(fig)
    return float(gains.mean()),float(stops.mean())

def plot_discount(n_sessions,steps,patience,seed):
    ds=np.array([0.,.2,.35,.5,.7,.85,.95])
    values=[]
    for i,d in enumerate(ds):
        g,_=run_sessions(.15,4e-6,float(d),n_sessions,steps,patience,seed+i*10000)
        values.append(g.mean())
    plt.figure(figsize=(7,4.6));plt.plot(ds,values,'-o');plt.xlabel('Discount factor δ')
    plt.ylabel('Normalized profit gain Δ');plt.title('Figure 3 analogue: Discount factor')
    plt.tight_layout();plt.savefig(OUT/'fig3_discount.png',dpi=170);plt.close()
    return ds,values

def plot_grid(n_sessions,steps,patience,seed,grid_n):
    alphas=np.linspace(.025,.25,grid_n); betas=np.linspace(.3e-6,2e-5,grid_n)
    z=np.zeros((grid_n,grid_n))
    for i,a in enumerate(alphas):
        for j,b in enumerate(betas):
            g,_=run_sessions(float(a),float(b),.95,n_sessions,steps,patience,seed+i*1000+j*31)
            z[i,j]=g.mean()
    plt.figure(figsize=(7,5));plt.imshow(z,origin='lower',aspect='auto',extent=[betas[0]*1e5,betas[-1]*1e5,alphas[0],alphas[-1]],cmap='OrRd')
    plt.colorbar(label='Normalized profit gain Δ');plt.xlabel('β × 10⁵');plt.ylabel('α')
    plt.title('Figure 1 analogue: Learning/exploration grid')
    plt.tight_layout();plt.savefig(OUT/'fig1_heatmap.png',dpi=170);plt.close()
    return alphas,betas,z

def main():
    p=argparse.ArgumentParser();p.add_argument('--mode',choices=['quick','paper'],default='quick')
    p.add_argument('--seed',type=int,default=2026);p.add_argument('--sessions',type=int)
    p.add_argument('--steps',type=int);p.add_argument('--grid',type=int)
    args=p.parse_args()
    if args.mode=='quick':
        sessions=args.sessions or 5;steps=args.steps or 450_000;grid=args.grid or 4;patience=100_000
    else:
        sessions=args.sessions or 1000;steps=args.steps or 10_000_000;grid=args.grid or 100;patience=100_000
    print(f'Nash price={p_n:.6f}, monopoly price={p_m:.6f}, Nash profit={pi_n:.6f}, monopoly profit={pi_m:.6f}',flush=True)
    print(f'Mode={args.mode}; sessions={sessions}, steps cap={steps}, grid={grid}',flush=True)
    avg,stops=plot_main(.15,4e-6,sessions,steps,patience,args.seed)
    print('Baseline normalized gain:',round(avg,4),'avg iterations:',round(stops),flush=True)
    ds,vals=plot_discount(max(2,sessions//2),steps,patience,args.seed+200000)
    print('Discount results:',list(zip(ds,np.round(vals,3))),flush=True)
    al,be,z=plot_grid(max(2,sessions//2),steps,patience,args.seed+400000,grid)
    import csv
    with open(OUT/'summary.csv','w',newline='') as f:
        w=csv.writer(f);w.writerow(['metric','value']);w.writerow(['p_nash',p_n]);w.writerow(['p_monopoly',p_m]);w.writerow(['pi_nash',pi_n]);w.writerow(['pi_monopoly',pi_m]);w.writerow(['baseline_mean_gain',avg]);w.writerow(['baseline_mean_training_steps',stops]);w.writerow(['mode',args.mode]);w.writerow(['sessions',sessions]);w.writerow(['max_steps',steps]);w.writerow(['grid_n',grid])
        for d,v in zip(ds,vals):w.writerow([f'discount_{d}',v])
    print('Saved figures and summary to',OUT,flush=True)
if __name__=='__main__':main()
